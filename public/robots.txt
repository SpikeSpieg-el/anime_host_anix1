# ВАЖНО: здесь НЕТ отдельной группы `User-agent: Googlebot` / `YandexBot`,
# и это сделано намеренно, а не по недосмотру.
#
# По спецификации Google группы с конкретным user-agent и группа `*`
# НЕ объединяются: "User agent specific groups and global groups (*) are not
# combined" (developers.google.com/crawling/docs/robots-txt/robots-txt-spec).
# Значит, отдельная группа Googlebot должна была бы повторять ВСЕ правила
# ниже целиком — иначе краулер потерял бы запреты на /admin/, /bookmarks и
# т.д. Любая последующая правка об этом забудет.
#
# Плюс такая группа почти наверняка вскрыла бы /embed/ для Googlebot, а это
# ровно тот вектор, из-за которого прилетали страйки: Googlebot пошёл бы по
# нашим ссылкам на Kodik и связал домен с источником. Поэтому плеер закрыт
# для ВСЕХ краулеров без исключения, включая поисковые.
#
# Текущая группа `*` уже даёт краулерам всё необходимое: обход страниц,
# CSS/JS-бандлы и доступ к превью через /api/thumbnail-proxy.
User-agent: *
Allow: /

# Закрытые разделы
Disallow: /admin/
Disallow: /18+/
Disallow: /private/

# ВАЖНО: /_next/ НЕ закрываем.
# Здесь раньше стоял `Disallow: /_next/`. Он блокировал CSS и JS-бандлы
# (/ _next/static) и оптимизированные картинки (/_next/image), из-за чего
# Googlebot не мог отрендерить страницу: видел неоформленную разметку без
# картинок. Google прямо просит не закрывать CSS/JS — это ухудшает и оценку
# mobile-friendly, и качество индексации, и Core Web Vitals.

# Страница плеера. Ссылки вида /embed/<токен> живут считаные минуты
# (AES-GCM-токен со сроком), отдают X-Robots-Tag: noindex и 404 для роботов
# (см. app/embed/[token]/route.ts) — обходить их бессмысленно, только расход
# краулингового бюджета. Поэтому Disallow оставляем.
# ВАЖНО: VideoObject.embedUrl на /watch больше НЕ ссылается сюда — раньше там
# был /embed/{id} без токена, который всегда отдавал 404 и ломал разметку видео.
Disallow: /embed/

# API routes - разрешаем только для изображений, остальные блокируем.
#
# Порядок важен и намеренный. Правило Allow должно идти ДО общего Disallow,
# иначе при беглой правке правил легко получить блокировку превью.
# Google при конфликте правил применяет НАИМЕНЕЕ ограничительное из них
# (developers.google.com/crawling/docs/robots-txt/robots-txt-spec, раздел
# "Order of precedence for rules"), поэтому /api/thumbnail-proxy остаётся
# доступным несмотря на Disallow: /api/.
#
# Это правило ОБЯЗАТЕЛЬНО: thumbnailUrl в VideoObject указывает именно сюда,
# и без доступа к нему Google отдаёт причину "Доступ к уменьшенным
# изображениям запрещен файлом robots.txt".
Allow: /api/thumbnail-proxy
Allow: /api/fallback-poster
Disallow: /api/

# Персональные страницы пользователей (не должны индексироваться)
Disallow: /bookmarks
Disallow: /history
Disallow: /settings
Disallow: /activate
Disallow: /test-search
Disallow: /market-dashboard

# Избегаем дубликатов контента
Disallow: /*?sort=
Disallow: /*?filter=
Disallow: /*?page=
Disallow: /*?search=

# Yandex: Clean param для борьбы с дубликатами URL-параметров
# Яндекс игнорирует Disallow: /*?param= по-другому, ему нужна Clean-param
#
# ВАЖНО: genre и year УБРАНЫ из этих списков. Они меняют содержимое страницы
# (например /catalog?genre=Исекай&year=<год> — это другая выдача и другой
# запрос), и
# страницы под них обязаны индексироваться с self-canonical. Когда genre стоял
# в Clean-param, Яндекс срезал параметр и видел только голый /catalog — то есть
# жанровые страницы для него просто не существовали.
#
# sort/page/search/q/filter и служебные status/kind/score остаются: их страницы
# закрыты от индексации (см. Disallow выше и canonical на /catalog).
Clean-param: sort&page&search&q /
Clean-param: sort&page&search&q&status&kind&score /catalog
Clean-param: episode /watch
